  B. A class named ElectronAdapter that implements StorageAdapter.
     Each method delegates to the window globals as follows.
     All window globals are accessed with the non-null assertion operator (!).

       saveCharacter(id, data)
         -> await window.characterStore!.saveCharacter(id, data)
         -> return void

       loadCharacter(id)
         -> return window.characterStore!.loadCharacter(id)

       listCharacters()
         -> return window.characterStore!.listCharacters()

       deleteCharacter(id)
         -> await window.characterStore!.deleteCharacter(id)
         -> return void

       readFile(filename)
         -> return window.equipmentStore!.readFile(filename)

         -> await window.equipmentStore!.writeFile(filename, content)
         -> return void

       fileExists(filename)
         -> return window.equipmentStore!.fileExists(filename)

  C. A class named CapacitorAdapter that implements StorageAdapter.

     At the top of the file, import from '@capacitor/filesystem':
       import { Filesystem, Directory, Encoding } from '@capacitor/filesystem'

     The class uses two module-level string constants (define outside the class):
       const CHAR_DIR  = 'characters'
       const EQUIP_DIR = 'equipment'

     All Filesystem calls use:
       directory: Directory.Data
       encoding:  Encoding.UTF8   (for read/write file operations only)

     The class has a private method:
       private async ensureDir(dir: string): Promise<void>
         Calls Filesystem.mkdir with:
           path:      dir
           directory: Directory.Data
           recursive: true
         Wraps the call in try/catch. On any error, does nothing (directory
         likely already exists).

     Method implementations:

       saveCharacter(id, data)
         1. await this.ensureDir(CHAR_DIR)
         2. await Filesystem.writeFile({
              path:      CHAR_DIR + '/' + id + '.json',
              data:      JSON.stringify(data, null, 2),
              directory: Directory.Data,
              encoding:  Encoding.UTF8,
            })

       loadCharacter(id)
         1. try:
              const result = await Filesystem.readFile({
                path:      CHAR_DIR + '/' + id + '.json',
                directory: Directory.Data,
                encoding:  Encoding.UTF8,
              })
              return JSON.parse(result.data as string)
         2. catch any error: return null

       listCharacters()
         1. await this.ensureDir(CHAR_DIR)
         2. try:
              const result = await Filesystem.readdir({
                path:      CHAR_DIR,
                directory: Directory.Data,
              })
              Filter result.files:
                keep only entries where f.name ends with '.json'
                AND f.name does NOT start with '__'
              Map the filtered entries:
                for each f, return f.name with the last 5 characters ('.json') removed
              return the mapped array
         3. catch any error: return []

       deleteCharacter(id)
         1. try:
              await Filesystem.deleteFile({
                path:      CHAR_DIR + '/' + id + '.json',
                directory: Directory.Data,
              })
         2. catch any error: do nothing

       readFile(filename)
         1. try:
              const result = await Filesystem.readFile({
                path:      EQUIP_DIR + '/' + filename,
                directory: Directory.Data,
                encoding:  Encoding.UTF8,
              })
              return result.data as string
         2. catch any error: return null

       writeFile(filename, content)
         1. await this.ensureDir(EQUIP_DIR)
         2. await Filesystem.writeFile({
              path:      EQUIP_DIR + '/' + filename,
              data:      content,
              directory: Directory.Data,
              encoding:  Encoding.UTF8,
            })

       fileExists(filename)
         1. try:
              await Filesystem.stat({
                path:      EQUIP_DIR + '/' + filename,
                directory: Directory.Data,
              })
              return true
         2. catch any error: return false

  D. A module-level variable for memoization:
       let _adapter: StorageAdapter | null = null

  E. An exported function named getStorageAdapter():
       Return type: StorageAdapter
       Logic:
         if _adapter is null:
           Check: typeof (window as any).characterStore !== 'undefined'
           If true:  _adapter = new ElectronAdapter()
           If false: _adapter = new CapacitorAdapter()
         return _adapter

STEP 5 — REPLACE FILE: src/renderer/src/services/ipc.ts
------------------------------------------------------------------------
File: src/renderer/src/services/ipc.ts
Action: REPLACE entire file content.

Write the following content exactly:

------ BEGIN FILE CONTENT ------
import { getStorageAdapter } from './storageAdapter'
import type { Character } from '@/entities/character/types'

const CUSTOM_ITEMS_KEY = '__customItems__'

export const ipcService = {
  save: (id: string, character: Character): void => {
    getStorageAdapter().saveCharacter(id, character)
  },

  load: (id: string): Promise<Character | null> => {
    return getStorageAdapter().loadCharacter(id) as Promise<Character | null>
  },

  list: (): Promise<string[]> => {
    return getStorageAdapter().listCharacters()
  },

  delete: (id: string): void => {
    getStorageAdapter().deleteCharacter(id)
  },

  saveCustomItems: (items: Record<string, unknown>): void => {
    getStorageAdapter().saveCharacter(CUSTOM_ITEMS_KEY, items)
  },

  loadCustomItems: (): Promise<Record<string, unknown> | null> => {
    return getStorageAdapter().loadCharacter(CUSTOM_ITEMS_KEY) as Promise<Record<string, unknown> | null>
  },
}

export const equipmentIpc = {
  readFile: (filename: string): Promise<string | null> =>
    getStorageAdapter().readFile(filename),

  writeFile: (filename: string, content: string): Promise<{ ok: true }> =>
    getStorageAdapter().writeFile(filename, content).then(() => ({ ok: true as const })),

  fileExists: (filename: string): Promise<boolean> =>
    getStorageAdapter().fileExists(filename),
}
------ END FILE CONTENT ------
